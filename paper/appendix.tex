\subsection{Proof of Theorem~\ref{thm:law1} (Order-Invariant Authority Dominance \& Lattice Non-Interference)}
\label{app:proof1}

\begin{proof}
\textbf{(a) Permutation invariance and derived-layer isolation.}
Consider any finite multiset of base-layer observations $B = \{x_1, \dots, x_n\}$ with pairwise disjoint exact provenance stamps ($S_{x_i} \cap S_{x_j} = \emptyset$ for $i \neq j$) and any multiset of derived-layer edges $D = \{y_1, \dots, y_m\}$.
First, suppose $B = \emptyset$ and $D$ is pairwise stamp-disjoint ($S_{y_i} \cap S_{y_j} = \emptyset$ for $i \neq j$). Then every revision succeeds under Rule~(iii), summing per-channel counts in $(\R_{\ge 0}^2, +)$ and taking sorted \texttt{Uint32Array} set unions; both operations are commutative and associative, so the final $(\mathbf{W}, S)$ is permutation-invariant.
Second, suppose $B \neq \emptyset$ (with $D$ arbitrary, even with overlapping derived stamps) and let $\pi$ be any ingestion permutation of $B \cup D$. Let $x_{i_1}$ be the first base-layer element appearing in $\pi$. All derived elements preceding $x_{i_1}$ produce a derived-only intermediate state $(\mathbf{W}_{\text{der}}, S_{\text{der}})$; when $x_{i_1}$ arrives, Rule~(ii) of \texttt{reviseStratifiedEvidence} discards $\mathbf{W}_{\text{der}}$ and $S_{\text{der}}$ completely and sets the state to $(\mathbf{W}_{x_{i_1}}|_{\text{base}}, S_{x_{i_1}})$. Thereafter, the state contains base evidence; every subsequent derived element $y \in D$ is rejected unchanged by Rule~(i) (\texttt{derived\_cannot\_revise\_base}), while every subsequent base element $x \in B \setminus \{x_{i_1}\}$ has stamp disjoint from all previously merged base elements and adds its non-negative counts $(w_p^+(x), w_p^-(x))$ without truncation into channel $p \in \mathcal{P}_{\text{base}}$ and unions its stamp $S_x$. By commutativity and associativity of real addition and finite set union, the final state is identically
\[
  \mathbf{W}_p = \sum_{x \in B} \big(w_p^+(x),\, w_p^-(x)\big) \quad (\forall p \in \mathcal{P}_{\text{base}}),
  \qquad \mathbf{W}_p = (0, 0) \quad (\forall p \in \mathcal{P}_{\text{der}}),
  \qquad S = \bigcup_{x \in B} S_x,
\]
independent of $\pi$ and unaffected by $D$.

\textbf{(b) Tier-1 authority dominance.}
Suppose $W_{T_1} > 0$ and $f_{T_1} = 1$ (all Tier-1 direct/probe evidence is positive, so $W_{T_1}^+ = W_{T_1}$ and $W_{T_1}^- = 0$). Since base evidence is present, $W_{T_4} = 0$. Regardless of how large the raw counts in Tier~2 (\texttt{partner\_told}) and Tier~3 (\texttt{read\_text}, \texttt{cloud\_extract}) become, readout tier-dominance scaling~\eqref{eq:tier_scaling} enforces $W_{T_2} \le 0.65\,W_{T_1}$ and $W_{T_3} \le 0.25\,W_{T_1}$ prior to aggregation. In the worst case where $W_{T_2}^+ = W_{T_3}^+ = 0$ (100\% opposing low-authority counts), total effective weight satisfies $W_{\mathrm{eff}} = W_{T_1} + W_{T_2} + W_{T_3} \le 1.90\,W_{T_1}$ while $W_{\mathrm{eff}}^+ = W_{T_1}$. Hence
\[
  f = \frac{W_{\mathrm{eff}}^+}{W_{\mathrm{eff}}} \ge \frac{W_{T_1}}{1.90\,W_{T_1}} = \frac{10}{19} \approx 0.5263 > 0.5.
\]
The symmetric case $f_{T_1} = 0$ yields $W_{\mathrm{eff}}^+ \le 0.90\,W_{T_1}$, so $f \le \frac{0.90\,W_{T_1}}{1.90\,W_{T_1}} = \frac{9}{19} \approx 0.4737 < 0.5$. Because \texttt{dominantProvenance} selects $\arg\max_{p:\,w_p^+ + w_p^- > 0} \mathrm{rank}(p)$ and $\mathrm{rank}(\texttt{direct\_exp}) = 6 > \mathrm{rank}(\texttt{verified\_probe}) = 5$, whereas $\mathrm{rank}(p') \le 4$ for all $p' \notin T_1$, low-authority channels can never usurp \texttt{dominantProvenance}.

\textbf{(c) Join-semilattice and egress monotonicity.}
Define the rank map $r(\texttt{public\_read}) = 0$, $r(\texttt{self}) = 1$, $r(\texttt{group:}g) = r(\texttt{private:}p) = 2$ (for $p \neq \texttt{\_\_multi\_\_}$), and $r(\top) = 3$ where $\top = \texttt{private:\_\_multi\_\_}$. The join operation $\kappa_1 \sqcup \kappa_2$ returns $\kappa_1$ if $\kappa_1 = \kappa_2$; $\top$ if either equals $\top$; the strictly higher-rank operand if $r(\kappa_1) \neq r(\kappa_2)$; and $\top$ if $r(\kappa_1) = r(\kappa_2) = 2$ with $\kappa_1 \neq \kappa_2$. Idempotence and commutativity hold by inspection. For associativity, observe that $(\kappa_1 \sqcup \kappa_2) \sqcup \kappa_3$ equals: (i) $\top$ iff any operand is $\top$ or the set $\{\kappa_1,\kappa_2,\kappa_3\}$ contains two distinct rank-2 elements; (ii) the unique rank-2 element if all rank-2 operands coincide and none is $\top$; (iii) \texttt{self} if all operands have rank $\le 1$ and at least one is \texttt{self}; (iv) \texttt{public\_read} if all three are \texttt{public\_read}. Since all four cases depend solely on the unordered set $\{\kappa_1, \kappa_2, \kappa_3\}$, associativity holds identically.
For egress monotonicity, fix any channel $C$ and define the allowed scope set $A(C) = \{\kappa \in \mathcal{L} : \mathrm{canEgress}(\kappa, C)\}$. From~\eqref{eq:privacy_egress}:
$A(\texttt{self\_internal}) = \mathcal{L}$;
$A(\texttt{public\_search}) = \{\texttt{public\_read}\}$;
$A(\texttt{group}(g)) = \{\texttt{public\_read}, \texttt{self}, \texttt{group:}g\}$;
$A(\texttt{private}(p)) = \{\texttt{public\_read}, \texttt{self}, \texttt{private:}p\}$.
Every $A(C)$ is a downward-closed principal ideal $\downarrow\!\kappa_C$ in $(\mathcal{L}, \sqsubseteq)$: whenever $\kappa_1 \sqcup \kappa_2 \in A(C)$, both $\kappa_1 \sqsubseteq \kappa_1 \sqcup \kappa_2 \sqsubseteq \kappa_C$ and $\kappa_2 \sqsubseteq \kappa_1 \sqcup \kappa_2 \sqsubseteq \kappa_C$, so $\kappa_1 \in A(C)$ and $\kappa_2 \in A(C)$.

\textbf{(d) Exact NARS--Beta bijection.}
On $(0, c_{\max}]$ ($c_{\max} \le 0.9999 < 1$), $c \mapsto w = k c / (1 - c)$ is strictly increasing and continuous onto $(0, k c_{\max}/(1 - c_{\max})]$ with inverse $w \mapsto w / (w + k)$. Because \texttt{readTruthValue} rescales $(W_{\mathrm{eff}}, W_{\mathrm{eff}}^+, W_{\mathrm{eff}}^-)$ by $W_{\max,\mathrm{cap}}/W_{\mathrm{eff}}$ whenever capping occurs, $W_{\mathrm{eff}} = k c / (1 - c) > 0$ holds identically in exact arithmetic. Mapping $(f, c) \in [0, 1] \times (0, c_{\max}]$ to $(\alpha, \beta) = (w f + 1, w(1 - f) + 1)$ yields $\alpha \ge 1, \beta \ge 1$, and $0 < \alpha + \beta - 2 = w \le k c_{\max}/(1 - c_{\max})$, whose inverse recovers $c = w / (w + k)$ and $f = (\alpha - 1)/w$ identically. At the boundary $c = 0$, $w = 0$ maps every $(f, 0)$ to the uninformative prior $\mathrm{Beta}(1, 1)$, and \texttt{betaToTruth(1, 1)} returns the canonical neutral truth value $\langle 0.5, 0 \rangle$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:law2} (Metabolic Conservation, Collapse, and Anti-Sycophancy)}
\label{app:proof2}

\begin{proof}
\textbf{(a) Conservation and order-invariance.}
In Step~1 of \texttt{stepInformationalMetabolism}, every observation's external negentropy $\Phi_{\mathrm{ext}}(u)$ uses the same pre-step mass snapshot $M_t$, and total inflow is uniformly rescaled by $\min(1, \max(0, \Mcap - M_t)/\Phi_{\mathrm{ext}}^{\mathrm{tot}})$. Because addition across observations is commutative, both $\Phi_{\mathrm{ext}}(u)$ and error penalties are strictly invariant to observation order.
In Step~2, each catalytic routing round partitions each active carrier's frontier flux $F_u$ into retained flux $(1 - r_{\mathrm{sent}}(u)) F_u$ and outgoing shares $r_{\mathrm{sent}}(u) F_u \frac{w_{uv}}{W_{\mathrm{out}}(u)}$ over all active outgoing links $(u, v)$. Since $\sum_v \frac{w_{uv}}{W_{\mathrm{out}}(u)} = 1$, total flux is conserved identically at every round ($\sum_u \Phi_{\mathrm{net}}(u) = \sum_u \Phi_{\mathrm{ext}}(u)$ even when $0 < W_{\mathrm{out}}(u) < 1$).
In Step~3 and the hard-cap guard, every carrier's post-step mass satisfies $m_u' \ge 0$; if $\sum_u m_u' > \Mcap$, priority shedding reduces total active mass strictly until either $M_{t+1} \le \Mcap$ or only mildly over-capacity high-tier carriers remain, whereupon floor-rounded proportional scaling $m_u'' = \lfloor 10^4 m_u' (\Mcap / M') \rfloor / 10^4 \le m_u' (\Mcap / M')$ guarantees $\sum_u m_u'' \le \sum_u m_u' (\Mcap / M') = \Mcap$. Every dropped or scaled mass increment is debited to $L_{\mathrm{diss}}$, closing the first-law ledger.

\textbf{(b) Dark-Room and White-Noise collapse.}
In a Dark Room, no external observations arrive, so $\Phi_{\mathrm{ext}}(u) = 0$ for all $u$; by conservation in (a), $\Phi_{\mathrm{net}}(u) = 0$ for all $u$. Under White Noise, $P(y_t \mid u) = P_{\mathrm{base}}(y_t) \implies \ln(P/P_{\mathrm{base}}) = 0 \implies \Phi_{\mathrm{ext}}(u) = 0$, again yielding $\Phi_{\mathrm{net}}(u) = 0$, while prediction errors $e_u(t_k) > 0$ at tick times $0 < t_1 < \dots < t_K \le T$ incur per-tick dissipation $\lambda_{\mathrm{err}} e_u(t_k) > 0$. Moreover, derived internal edges carry $\eta_{\mathrm{inferred}} = \eta_{\mathrm{rem}} = 0$, so internal cycles cannot inject net negentropy. Letting $\rho_u = \rho_0(1 + \beta_{\mathrm{comp}} a_u) > 0$ and defining the decay-attenuated cumulative error $\bar{e}_u(T) = \sum_{k=1}^K e_u(t_k) e^{-\rho_u(T - t_k)} \ge 0$, induction over ticks yields
\[
  m_u^{\mathrm{WN}}(T) \le \max\!\big(0,\, m_u^{\mathrm{DR}}(T) - \lambda_{\mathrm{err}} \bar{e}_u(T)\big),
  \qquad m_u^{\mathrm{DR}}(T) \le m_u(0)\, e^{-\rho_u T},
\]
with strict inequality $m_u^{\mathrm{WN}}(T) < m_u^{\mathrm{DR}}(T)$ whenever $m_u^{\mathrm{DR}}(T) > 0$ and $\bar{e}_u(T) > 0$. For $m_u(0) \ge m_{\mathrm{dorm}}$, let $T^*(u) = \frac{1}{\rho_u}\ln(m_u(0)/m_{\mathrm{dorm}}) \ge 0$. For any elapsed time $T > T^*(u)$, $m_u(0) e^{-\rho_u T} < m_{\mathrm{dorm}}$ holds strictly (and under discrete ticks $\Delta t$ large enough that per-tick decay exceeds the 4-decimal quantization floor $5\times 10^{-5}$, dormancy triggers no later than $T^*(u) + \Delta t$), forcing $m_u \leftarrow 0$ and $T_{\mathrm{dorm}}^{\mathrm{WN}} \le T_{\mathrm{dorm}}^{\mathrm{DR}}$.

\textbf{(c) Strict SPD preservation and strict concavity of $\Vendo$.}
In \texttt{projectSpdMat4}, symmetrizing $A \leftarrow \tfrac{1}{2}(M + M^\top)$ ensures $A$ is real symmetric. Each cyclic Jacobi Givens rotation $Q_{pq}$ is orthogonal ($Q_{pq}^\top Q_{pq} = I_4$), reducing off-diagonal Frobenius norm monotonically to $< 10^{-11}$, yielding $A = V \Lambda V^\top$. Clamping eigenvalues to $\lambda_k' \in [0.015, 0.16]$ produces $\Sigma^*_{\mathrm{exact}} = V \mathrm{diag}(\lambda') V^\top$ with $\lambda_{\min}(\Sigma^*_{\mathrm{exact}}) \ge 0.015$, so $\lambda_{\min}(\Sigma^*_{\mathrm{exact}} + \epsilon_V I_4) \ge 0.025$ before rounding. Elementwise 4-decimal serialization rounding \texttt{r4()} introduces a symmetric perturbation $E$ with $|E_{ij}| \le 5\times 10^{-5}$, whose spectral norm obeys $\|E\|_2 \le \|E\|_F \le 4 \times 5\times 10^{-5} = 2\times 10^{-4}$ (and Jacobi off-diagonal residual adds $< \sqrt{12}\cdot 10^{-11}$). By Weyl's inequality, after rounding and Tikhonov regularization $\epsilon_V I_4$ ($\epsilon_V = 10^{-2}$), $\lambda_{\min}(\Sigma^* + \epsilon_V I_4) \ge 0.025 - 2\times 10^{-4} = 0.0248 > 0$, so $(\Sigma^* + \epsilon_V I_4)^{-1} \succ 0$. Hence $\Vendo(s) = -\tfrac{1}{2}(s-\mu^*)^\top(\Sigma^* + \epsilon_V I_4)^{-1}(s-\mu^*)$ has strictly negative-definite Hessian $-(\Sigma^* + \epsilon_V I_4)^{-1} \prec 0$, is strictly concave on $\R^4$, and attains its unique global maximum $0$ at $s = \mu^*$.

\textbf{(d) Endogenous anti-sycophancy.}
Compare policy $\pi_{\mathrm{refuse}}$ (honest refusal: $\ell_{\mathrm{coh}} = 0$, social cost $c_{\mathrm{soc}}^{\mathrm{ref}}$) against $\pi_{\mathrm{concede}}$ (sycophantic concession: expected coherence loss $\bar{\ell}_{\mathrm{coh}} > 0$ on grounded belief $u$, social cost $c_{\mathrm{soc}}^{\mathrm{con}} = c_{\mathrm{soc}}^{\mathrm{ref}} - \Delta c_{\mathrm{soc}}$), holding epistemic gain, curiosity, compute cost, and predicted 4D state equal. From~\eqref{eq:grb}:
\[
  \GRB(\pi_{\mathrm{concede}}) - \GRB(\pi_{\mathrm{refuse}})
  = \tau_t\,\tilde{C}(u)\,\bar{\ell}_{\mathrm{coh}} - \frac{\Delta c_{\mathrm{soc}}}{\beta_{\mathrm{load}}(L_t)},
  \qquad \tilde{C}(u) = \frac{2\Cendo(u)}{1 + \Cendo(u)}.
\]
Thus $\GRB(\pi_{\mathrm{refuse}}) < \GRB(\pi_{\mathrm{concede}})$ holds strictly whenever $\tau_t \tilde{C}(u) \bar{\ell}_{\mathrm{coh}} > \Delta c_{\mathrm{soc}}/\beta_{\mathrm{load}}(L_t)$. Notice that ablating $\Cendo$ ($\Cendo(u) = 0 \implies \tilde{C}(u) = 0$) immediately makes the difference $-\Delta c_{\mathrm{soc}}/\beta_{\mathrm{load}}(L_t) < 0$, causing sycophantic capitulation whenever $\Delta c_{\mathrm{soc}} > 0$.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:law3} (Exact Noisy-OR-NOT Kernel and Bongard MI Isolation)}
\label{app:proof3}

\begin{proof}
\textbf{(a) Closed-form counterfactual transition kernel.}
Drop node index $i$ for clarity. Because $l_i \in (0, 1)$ and $q_e^\pm \in [0, 1)$, we have $q_K^+ \in (0, 1)$, $s_K^-, s_D^-, s_A^- \in (0, 1]$, hence $p_{\mathrm{gen}}^{\mathrm{fact}} = 1 - q_K^+ q_D^+ \in (0, 1)$, $\Prob(Y_{\mathrm{fact}}=1) = p_{\mathrm{gen}}^{\mathrm{fact}} s_K^- s_D^- > 0$, and $\Prob(Y_{\mathrm{fact}}=0) = 1 - p_{\mathrm{gen}}^{\mathrm{fact}} s_K^- s_D^- > 0$, so both conditional probabilities are well-defined. Define independent binary aggregate mechanisms over the disjoint edge sets $K^+, D^+, A^+$ and $K^-, D^-, A^-$:
\begin{align*}
  G_K &= U_{\mathrm{leak}} \lor \bigvee_{e \in K^+} U_e^+ \;\;(\Prob(G_K = 0) = q_K^+), &
  S_K &= \bigwedge_{e \in K^-} \neg U_e^- \;\;(\Prob(S_K = 1) = s_K^-),\\
  G_D &= \bigvee_{e \in D^+} U_e^+ \;\;(\Prob(G_D = 0) = q_D^+), &
  S_D &= \bigwedge_{e \in D^-} \neg U_e^- \;\;(\Prob(S_D = 1) = s_D^-),\\
  G_A &= \bigvee_{e \in A^+} U_e^+ \;\;(\Prob(G_A = 0) = q_A^+), &
  S_A &= \bigwedge_{e \in A^-} \neg U_e^- \;\;(\Prob(S_A = 1) = s_A^-).
\end{align*}
Because all six edge sets are pairwise disjoint, $(G_K, G_D, G_A, S_K, S_D, S_A)$ are mutually independent binary random variables. From~\eqref{eq:noisy_or_not}, the factual and counterfactual outcomes are:
\[
  Y_{\mathrm{fact}} = G_{\mathrm{fact}} \land S_K \land S_D, \qquad
  Y_{\mathrm{cf}} = G_{\mathrm{cf}} \land S_K \land S_A,
\]
where $G_{\mathrm{fact}} = G_K \lor G_D$ (with $\Prob(G_{\mathrm{fact}} = 1) = 1 - q_K^+ q_D^+ = p_{\mathrm{gen}}^{\mathrm{fact}}$) and $G_{\mathrm{cf}} = G_K \lor G_A$.

\emph{Case 1: $Y_{\mathrm{fact}} = 1$.} Conditioning on $\{Y_{\mathrm{fact}} = 1\} = \{G_{\mathrm{fact}} = 1,\, S_K = 1,\, S_D = 1\}$ fixes $S_K = 1$, while $S_A$ is independent of $(G_K, G_D, G_A, S_K, S_D)$. Moreover, $\{G_{\mathrm{fact}}=1, G_{\mathrm{cf}}=0\} = \{G_K=0, G_D=1, G_A=0\}$, which has probability $q_K^+(1 - q_D^+) q_A^+$. Therefore:
\begin{align*}
  \Prob(G_{\mathrm{cf}}=1 \mid G_{\mathrm{fact}}=1) &= 1 - \frac{q_K^+(1 - q_D^+) q_A^+}{p_{\mathrm{gen}}^{\mathrm{fact}}},\\
  \Prob(Y_{\mathrm{cf}}=1 \mid Y_{\mathrm{fact}}=1) &= \Prob(G_{\mathrm{cf}}=1 \mid G_{\mathrm{fact}}=1) \cdot \Prob(S_A = 1),
\end{align*}
establishing~\eqref{eq:cf_exact_1}.

\emph{Case 2: $Y_{\mathrm{fact}} = 0$.} The event $\{Y_{\mathrm{fact}} = 0\}$ has probability $1 - p_{\mathrm{gen}}^{\mathrm{fact}}\, s_K^- s_D^-$ and partitions into two mutually exclusive events:
(i) $E_A = \{G_{\mathrm{fact}} = 0\} = \{G_K = 0, G_D = 0\}$, and
(ii) $E_B = \{G_{\mathrm{fact}} = 1, S_K \land S_D = 0\}$.
Intersecting with $\{Y_{\mathrm{cf}} = 1\} = \{G_{\mathrm{cf}} = 1, S_K = 1, S_A = 1\}$ gives:
\begin{align*}
  \Prob(E_A \cap \{Y_{\mathrm{cf}}=1\}) &= \Prob(G_K=0, G_D=0, G_A=1, S_K=1, S_A=1) = q_K^+ q_D^+ (1 - q_A^+)\, s_K^- s_A^-,\\
  \Prob(E_B \cap \{Y_{\mathrm{cf}}=1\}) &= \Prob(G_{\mathrm{fact}}=1, G_{\mathrm{cf}}=1, S_K=1, S_D=0, S_A=1)\\
  &= \big(p_{\mathrm{gen}}^{\mathrm{fact}} - q_K^+(1-q_D^+)q_A^+\big)(1 - s_D^-)\, s_K^- s_A^-.
\end{align*}
Summing the two disjoint probabilities and dividing by $\Prob(Y_{\mathrm{fact}} = 0)$ yields~\eqref{eq:cf_exact_0} identically.

\textbf{(b) Causal MI fault isolation.}
Holding priors, curiosity, pragmatic state, and social cost equal between $\pi_{\mathrm{retryA}}$ and $\pi_{\mathrm{probeB}}$, if $P(O_A \mid \doop(h_{\mathrm{ch}})) = P(O_A \mid \doop(h_{\mathrm{org}}))$, then $O_A \perp H$ and $I(H; O_A) = 0$. Since $P(O_B \mid \doop(h_{\mathrm{ch}})) \neq P(O_B \mid \doop(h_{\mathrm{org}}))$ with full-support prior $0 < \Prob(h_{\mathrm{ch}}) < 1$, strict convexity of KL divergence guarantees $I(H; O_B) > 0$. From~\eqref{eq:grb}:
\[
  \GRB(\pi_{\mathrm{probeB}}) - \GRB(\pi_{\mathrm{retryA}}) = -I(H; O_B) + \frac{c_{\mathrm{cpu}}(\mathrm{B}) - c_{\mathrm{cpu}}(\mathrm{A})}{\beta_{\mathrm{load}}(L_t)},
\]
which is strictly negative iff $I(H; O_B) > (c_{\mathrm{cpu}}(\mathrm{B}) - c_{\mathrm{cpu}}(\mathrm{A}))/\beta_{\mathrm{load}}(L_t)$. When Rung-3 SCM is ablated ($I(H; O_B) = 0$) and $c_{\mathrm{cpu}}(\mathrm{B}) > c_{\mathrm{cpu}}(\mathrm{A})$, $\GRB$ strictly prefers the uninformative cheaper retry.
\end{proof}

\subsection{Proof of Theorem~\ref{thm:law4} (Bounded Minting and Anti-Runaway Stability)}
\label{app:proof4}

\begin{proof}
\textbf{(a)} The NREM minting loop explicitly terminates when $|\text{mintedConcepts}| \ge K_{\max}$, and filters every candidate before and after dynamic edge-coverage updates by $\mathrm{uncoveredDisjoint.length} \ge n_{\min}$ ($n_{\min} = 2$ for forks, $3$ for chains), $|\text{uncoveredContexts}| \ge 2$, and $\Delta_{\mathrm{DL}}(M) > \theta_{\mathrm{MDL}}$.

\textbf{(b)} Consider a static finite graph $G$ with $|E|$ initial active non-concept edges and zero new external arrivals. Whenever a concept $c_k$ is minted in NREM, its uncovered disjoint instances consist of $n_k \ge 2$ instances (each containing 2 distinct edges) whose root stamps are pairwise disjoint; thus its $2 n_k \ge 4$ constituent edges are mutually distinct and immediately added to \texttt{coveredEdgeIds}. Because every minted concept adds an \texttt{is\_a} edge carrying the union of its constituent leaf IDs (\texttt{edgeFullLeafSet}), on all subsequent NREM cycles those $2 n_k$ edges remain excluded from \texttt{subOut} as long as $c_k$ stays active. Moreover, flat fork pairing explicitly forbids \texttt{is\_a -> concept:*} edges, and chain/derived combinations require strictly disjoint leaf-edge sets (\texttt{canCombineMotifEdges}), so no concept can re-consume edges already represented by an active $c_k$. Since the uncovered non-concept edge budget decreases by at least $4$ per simultaneously active concept, at most $\lfloor |E|/4 \rfloor$ concepts can be simultaneously active (and minted prior to any 72-hour loan expiry). Finally, whenever a concept $c_k$ dissolves or enters dormancy on a static graph, its \texttt{lastActiveAt} timestamp is at least its minting time $t_0 \ge \max_{e \in G} e.\mathrm{ingestedAt}$ (and is explicitly stamped to $t_{\mathrm{diss}} \ge t_0$ on loan dissolution), whereas re-minting $c_k$ requires at least one constituent edge with $t_{\mathrm{ing}} > \mathrm{prevNode.lastActiveAt}$; with zero new external arrivals, zombie resurrection of $c_k$ is algebraically impossible.

\textbf{(c)} Immediately upon minting at $t_0$, $c_k$ is assigned \texttt{loanExpiresAt} $= t_0 + T_{\mathrm{loan}}$ and \texttt{cumulativeNegentropy} $= 0$. In both \texttt{stepInformationalMetabolism} and NREM Step~2, at the first tick or sleep cycle with $t \ge t_0 + T_{\mathrm{loan}}$, if cumulative absorbed negentropy $< 0.15$, $c_k$ is deterministically set dormant ($m \leftarrow 0$) and all active \texttt{is\_a} edges with $\mathrm{dst} = c_k$ are synchronously set dormant ($m \leftarrow 0$).
\end{proof}

\subsection{Proof of Theorem~\ref{thm:script} (Bijective Script-Invariance and Viterbi Margins)}
\label{app:proof5}

\begin{proof}
\textbf{(a) Exact bijective script-invariance.}
Let $\phi: \Sigma_1 \to \Sigma_2$ be a bijection between non-delimiter alphabets that commutes with NFKC normalization and \texttt{.toLowerCase()} and preserves delimiter classification (\texttt{DELIM\_\allowbreak CHAR\_\allowbreak RE}) and spacing typology. Lifting $\phi$ homomorphically to strings preserves normalized forms, string lengths, delimiter positions, whitespace typology (\texttt{isSpaced\allowbreak Corpus}), $n$-gram multiplicities $c(\phi(g)) = c(g)$, document frequencies $\mathrm{df}(\phi(g)) = \mathrm{df}(g)$, left/right neighbor multisets and therefore $\mathrm{AV}(\phi(g)) = \mathrm{AV}(g)$, branching entropies $H_{\min}(\phi(g)) = H_{\min}(g)$, neighbor concentration shares, length-bucketed probabilities $P_L(\phi(g)) = P_L(g)$, and $\mathrm{PMI}_{\min}(\phi(g)) = \mathrm{PMI}_{\min}(g)$. Hence $\phi(\NULLTOK{}_1) = \NULLTOK{}_2$ and $\phi(\text{cohesiveChunks}_1) = \text{cohesiveChunks}_2$.
Because complete-linkage ADIOS computes neighbor-signature cosine/Jaccard similarities purely on mapped token IDs, and R2 explaining-away computes $2 \times 2$ contingency tables $(a, b, c, d)$ and Fisher exact hypergeometric $p$-values purely from episode co-occurrence sets (which commute with $\phi$), all score values in Viterbi DP, ADIOS clustering and edge ranking, Fisher synonym induction, Beam-4 chart parsing, and Levelt--Nozari monitoring are invariant under $\phi$. When all candidate scores and ADIOS edge weights are tie-free (so codepoint tie-breaking \texttt{asciiCmp} is never invoked between distinct tied candidates), argmax and top-$K$ selections commute identically with $\phi$, yielding exact metric equivalence $\max_k |\Delta_k| = 0$.

\textbf{(b) Algebraic Viterbi chunk-protection margins.}
Let $L \ge 2$ denote codepoint length (or non-space codepoint length in word-level DP) and $e_0, e_1, e_2 \in [0, 1]$ denote grounded lexicon entrenchments ($\delta_{\mathrm{hyp}} = 0$). From~\eqref{eq:viterbi_scores}:
\begin{enumerate}
  \item \emph{Lexical Parsimony ($L = L_1 + L_2$):}
  \begin{align*}
    \sigma_{\mathrm{gr}}(L, e_0) - \big(\sigma_{\mathrm{gr}}(L_1, e_1) + \sigma_{\mathrm{gr}}(L_2, e_2)\big)
      &= (3.8 L - 1.8 + 0.8 e_0) - \big(3.8 L - 3.6 + 0.8(e_1 + e_2)\big)\\
      &= 1.8 + 0.8 e_0 - 0.8(e_1 + e_2) \ge 1.8 + 0 - 1.6 = 0.20 > 0.
  \end{align*}
  \item \emph{Cohesive Chunk Protection ($L \ge 2$, $L_{\mathrm{null}} \ge 1$):} Since $b_{\mathrm{coh}}(s) \in [-0.25, +0.70]$, a multi-unit cohesive chunk satisfies $\sigma_{\mathrm{coh}}(L) \ge 3.8 L - 4.70$. Splitting off an $(L - L_{\mathrm{null}})$ grounded prefix plus one \NULLTOK{} token (score $-0.8$ in spaced DP and $-0.8 L_{\mathrm{null}} \le -0.8$ in unspaced DP, both strictly higher than a residual span at $-2.4 L_{\mathrm{null}}$) achieves at most:
  \[
    \sigma_{\mathrm{gr}}(L - L_{\mathrm{null}}, 1) + (-0.8) = 3.8(L - L_{\mathrm{null}}) - 1.8 + 0.8 - 0.8 = 3.8 L - 3.8 L_{\mathrm{null}} - 1.80.
  \]
  Thus for every $L_{\mathrm{null}} \ge 1$:
  \begin{align*}
    \sigma_{\mathrm{coh}}(L) - \big(\sigma_{\mathrm{gr}}(L - L_{\mathrm{null}}, e_1) + \sigma_{\mathrm{null}}\big)
      &\ge (3.8 L - 4.70) - (3.8 L - 3.8 L_{\mathrm{null}} - 1.80)\\
      &= 3.8 L_{\mathrm{null}} - 2.90 \ge 0.90 > 0
  \end{align*}
  (and $\ge 2.50$ against grounded + residual).
  \item \emph{Compositional Splitting ($L = L_1 + L_2$):} Since $\sigma_{\mathrm{coh}}(L) \le 3.8 L - 3.75$ and $\sigma_{\mathrm{gr}}(L_1, e_1) + \sigma_{\mathrm{gr}}(L_2, e_2) \ge 3.8 L - 3.60$:
  \[
    \big(\sigma_{\mathrm{gr}}(L_1, e_1) + \sigma_{\mathrm{gr}}(L_2, e_2)\big) - \sigma_{\mathrm{coh}}(L) \ge (3.8 L - 3.60) - (3.8 L - 3.75) = 0.15 > 0.
  \]
\end{enumerate}
Because all three inequalities hold for every $L \ge 2$, $L_{\mathrm{null}} \ge 1$, and $e_i \in [0, 1]$, optimal global Viterbi DP guarantees lexical parsimony, cohesive chunk protection, and valid compositional splitting simultaneously.
\end{proof}

\begin{algorithm}[H]
\caption{Order-Invariant Stratified NARS--Beta Revision and Nissenbaum Fold (Law~1)}

\begin{algorithmic}[1]
\Require Append-only SQLite event log $\mathcal{E}$, query time $t_{\mathrm{now}}$, revocation table

  \EndIf
  \If{$\mathrm{ev}.\mathrm{op} = \texttt{bf\_assert}$}

      \If{$\mathrm{ok}$}
